} if (parts[0] === "embed" && parts[1]) return url; } if (host === "youtu.be") { const id = u.pathname.split("/").filter(Boolean)[0]; if (id) return `https://www.youtube.com/embed/${encodeURIComponent(id)}?enablejsapi=1`;
